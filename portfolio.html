        document.addEventListener("DOMContentLoaded", function() {

            // 관찰할 대상 요소들 (가로 막대, 세로 막대)

            const bars = document.querySelectorAll('.bar-fill-prev, .bar-fill-curr, .v-bar-total, .v-bar-obsolete, .v-bar-cost, .v-bar-sales');

            

            const observer = new IntersectionObserver((entries) => {

                entries.forEach(entry => {

                    if (entry.isIntersecting) {

                        const target = entry.target;

                        

                        // HTML에 심어둔 data-width, data-height 값을 가져옴

                        const targetWidth = target.getAttribute('data-width');

                        const targetHeight = target.getAttribute('data-height');


                        // 가로 막대 애니메이션 트리거

                        if (target.classList.contains('bar-fill-prev')) {

                            target.style.width = targetWidth;

                            target.style.animation = 'growBar 1.2s ease-out forwards';

                        }

                        if (target.classList.contains('bar-fill-curr')) {

                            target.style.width = targetWidth;

                            target.style.animation = 'growBar 1.2s ease-out 0.2s forwards'; // 0.2초 딜레이

                        }


                        // 세로 막대 애니메이션 트리거 (Inventory & Promo)

                        if (target.classList.contains('v-bar-total') || target.classList.contains('v-bar-obsolete') || target.classList.contains('v-bar-cost') || target.classList.contains('v-bar-sales')) {

                            target.style.height = targetHeight;

                        }

                        

                        // 한 번 애니메이션 실행 후에는 관찰 중단 (성능 최적화)

                        observer.unobserve(target); 

                    }

                });

            }, { threshold: 0.2 }); // 요소의 20%가 보일 때 실행

        });
