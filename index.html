/*
[Table Schema Reference]
- customers: customer_id, membership_level, join_date
- orders: order_id, customer_id, order_date, total_paid
- order_items: order_id, sku_code, qty, price
- products: sku_code, category, brand_name
*/

-- [Step 1] 화장품 카테고리(K-Beauty)의 멤버십 등급별 장바구니 분석(UPT, AOV)
-- 다중 테이블 JOIN을 통해 고객 등급별 구매 행태를 구조화

WITH target_orders AS (
    SELECT 
        o.order_id,
        c.membership_level,
        o.total_paid,
        SUM(oi.qty) AS total_qty
    FROM orders o
    JOIN customers c ON o.customer_id = c.customer_id
    JOIN order_items oi ON o.order_id = oi.order_id
    JOIN products p ON oi.sku_code = p.sku_code
    WHERE p.category = 'K-Beauty'
      AND o.order_date BETWEEN '2025-01-01' AND '2025-06-30'
    GROUP BY o.order_id, c.membership_level, o.total_paid
)
SELECT 
    membership_level,
    COUNT(order_id) AS order_cnt,
    ROUND(AVG(total_qty), 1) AS avg_upt,  -- 객당 구매수량 (목표 지표 1)
    ROUND(AVG(total_paid), 0) AS avg_aov  -- 객단가 (목표 지표 2)
FROM target_orders
GROUP BY membership_level
ORDER BY avg_upt ASC;

개요 (Objective)

브랜드별·멤버십별로 파편화된 기존 할인 구조가 카테고 전체의 UPT(객당 구매 수량) 및 AOV(객단가) 개선에 한계를 보이는 상황에서, 고객 등급과 브랜드 가격대에 관계없이 확장 적용 가능한 구매 수량 기반 할인(PCS) 전략을 설계·검증하여 카테고리 핵심 KPI를 개선하고자 했습니다.

진행한 일 (Task)

배경

카테고리 매출 볼륨은 증가 추세였으나, UPT와 AOV는 2분기 연속 둔화되는 상반된 흐름을 보였습니다.

특히 저단가 브랜드에서 구매 수량 감소가 두드러졌으며, 이는 카테고리 전반의 성장 효율 저하로 이어지고 있었습니다.

문제 정의

기존 증정 프로모션은 고단가 브랜드 및 상위 멤버십 고객에게만 반응이 집중되는 경향이 있었고, 구매 수량 증가보다는 단일 상품 구매에 대한 보상 효과에 그쳐 카테고리 전체 개선으로는 연결되지 않았습니다.

해결 전략

UPT와 AOV를 동시에 개선하기 위해서는 ①특정 브랜드에 국한되지 않고 ②구매 수량 자체를 의사결정 변수로 만들며 ③멤버십 하위 고객일수록 체감 효과가 큰 프로모션 구조가 필요하다고 판단했습니다.

이에 따라 고객의 장바구니 구성 행동을 직접적으로 유도할 수 있는 PCS(구매 수량 기반 할인) 프로모션을 핵심 대안으로 설정했습니다.

성장한 점 (Growth)

로컬 접근성이 높은 한국 화장품 카테고리에서 멤버십 하위 고객 유입과 구매 전환이 구조적으로 어렵다는 한계를 확인했습니다. 그러나 PCS 할인이 멤버십 등급 간 허들을 낮추는 역할을 수행하며, 골드 멤버십 미만 고객의 평균 구매 수량을 유의미하게 끌어올리는 데 효과적이었습니다. 이를 통해 멤버십 등급 할인 외에도, 구매 행동을 기반으로 한 멤버십 상향 유도 레버리지를 새롭게 발견할 수 있었습니다.

상세 분석 과정 및 역량 보기
실무 분석 파이프라인 (Data Pipeline) SQL 
/*
[Table Schema Reference]
- brands: brand_id, brand_name
- products: sku_code, brand_id, unit_cost, retail_price
- inventory: sku_code, current_stock, last_inbound_date
*/

-- [Step 1] C브랜드의 1년 이상 장기 체화 재고(Obsolete Stock) 금액 및 비중 산출
-- 재고자산평가손익 관점의 리스크 모니터링 쿼리

WITH inventory_aging AS (
    SELECT 
        p.sku_code,
        i.current_stock,
        p.unit_cost,
        (i.current_stock * p.unit_cost) AS stock_value,
        DATEDIFF(CURRENT_DATE, i.last_inbound_date) AS aging_days
    FROM inventory i
    JOIN products p ON i.sku_code = p.sku_code
    JOIN brands b ON p.brand_id = b.brand_id
    WHERE b.brand_name = 'C_Brand' 
      AND i.current_stock > 0
)
SELECT 
    CASE 
        WHEN aging_days >= 365 THEN '1. Obsolete (365d+)'
        WHEN aging_days >= 180 THEN '2. Slow-moving (180d+)'
        ELSE '3. Healthy (<180d)'
    END AS stock_status,
    SUM(current_stock) AS total_qty,
    SUM(stock_value) AS total_value,
    ROUND(SUM(stock_value) / (SELECT SUM(stock_value) FROM inventory_aging) * 100, 1) AS value_portion_pct
FROM inventory_aging
GROUP BY stock_status
ORDER BY stock_status;

개요 (Objective)

코로나 기간 동안 누적된 장기 체화 재고를 전략적으로 소진해 묶여 있던 자금을 회수하고, 이를 베스트 상품 재투자로 연결함으로써 브랜드의 매출 성장 엔진을 재가동하고자 했습니다.

진행한 일 (Task)

배경

코로나 이전 면세 호황기에 대량 발주된 재고가 팬데믹 기간 동안 소진되지 못하며 장기 체화 재고로 전환되었습니다.

문제 정의

과도한 체화 재고가 매입 예산(OTB)을 잠식하며 트렌드 대응 상품 및 베스트 SKU 발주가 불가능한 상태였고, 이는 ‘신규 상품 부재 → 매출 둔화 → 재고 적체 심화’의 악순환 구조를 고착화시키고 있었습니다.

과정 (Process)

Action 1. 재고 구조 분석 및 타겟 설정
전체 재고를 SKU·입고 시점·회전율 기준으로 전수 분석하여 1년 이상 체화 재고를 핵심 관리 대상으로 정의했습니다. 향후 재고자산평가손익 관점에서 손실 가능성이 높은 악성 재고를 우선 소진 타겟으로 선정했습니다.
Action 2. 전략적 소진 (Exit Strategy)
총 8억 원의 프로모션 예산을 ‘비용’이 아닌 ‘자산 회수 수단’으로 활용해 약 21억 원 규모의 저효율 재고를 현금 자산으로 전환했습니다.
Action 3. 선순환 투자 (Re-investment)
확보된 유동성을 활용해 품절로 판매 기회를 상실하고 있던 베스트 SKU에 리오더를 실행, 상품 경쟁력을 회복시켰습니다.

거래선 다변화 기반의 구조적 이익 모델 전환 프로젝트

→ 특정 벤더 발주 중단 시 발생하던 비수기 매출 공백을 완화하고, 다수 거래선을 기반으로 한 공급 안정성 확보

/*
[Table Schema Reference]
- vendors: vendor_id, vendor_name, contract_type
- products: sku_code, brand_id, vendor_id
- sales_summary: sale_date, sku_code, net_revenue, cogs (Cost of Goods Sold)

-- [Step 1] M브랜드의 벤더별 매출 의존도 및 공헌이익률(CM%) 윈도우 함수 분석
-- 특정 벤더에 편중된 구조와 수익성 한계점 파악

WITH vendor_sales AS (
    SELECT 
        v.vendor_name,
        SUM(s.net_revenue) AS total_revenue,
        SUM(s.net_revenue - s.cogs) AS total_cm -- 공헌이익(매출 - 변동비)
    FROM sales_summary s
    JOIN products p ON s.sku_code = p.sku_code
    JOIN vendors v ON p.vendor_id = v.vendor_id
    WHERE p.brand_id = 'M_Brand'
      AND s.sale_date BETWEEN '2025-01-01' AND '2025-12-31'
    GROUP BY v.vendor_name
)
SELECT 
    vendor_name,
    total_revenue,
    ROUND(total_revenue / SUM(total_revenue) OVER () * 100, 1) AS revenue_share_pct,
    total_cm,
    ROUND((total_cm / total_revenue) * 100, 2) AS cm_margin_pct
FROM vendor_sales
ORDER BY total_revenue DESC;

개요 (Objective)

특정 거래선 의존도가 85%에 달해, 해당 벤더의 발주 중단 시 3분기 매출이 급감하는 구조적 취약점을 해결하고자 했습니다. 비수기 대체 거래선을 발굴하고 이를 고마진 구조로 정착시켜, 단기 매출 방어를 넘어 브랜드의 수익 체질을 개선하는 것을 목표로 설정했습니다.

진행한 일 (Task)

문제 정의

매출의 대부분이 특정 벤더(Vendor A)에 집중된 박리다매 구조로, 해당 벤더가 매입을 중단하는 3분기에는 대체 수단이 없어 브랜드 전체가 비수기 타격을 반복적으로 겪고 있었습니다.

해결 전략

초기에는 3분기 매출 방어를 위한 대체 거래선 발굴로 접근했습니다. 그러나 거래선 다변화 과정에서 협상력이 개선되고, 고마진 구조로의 전환 가능성이 확인되며 전략의 방향을 ‘비수기 방어’에서 ‘구조적 이익률 개선’으로 확장했습니다.

성장한 점 (Growth)

단기적인 비수기 매출 방어 활동이 거래 구조와 마진 구조를 재설계하는 계기가 될 수 있음을 경험했습니다. 매출 볼륨(Top-line)이 일부 감소하더라도, 마진율이 높은 건전한 구조(Bottom-line)로 전환하는 것이 장기적으로 더 큰 이익을 창출한다는 점을 데이터로 입증했습니다.
