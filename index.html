#01 PCS 할인 스킴 분석 SQL

-- 멤버십 등급별 UPT 및 AOV 산출 쿼리
SELECT 
    membership_level,
    COUNT(order_id) as total_orders,
    SUM(quantity) / COUNT(order_id) as UPT,
    SUM(amount) / COUNT(order_id) as AOV
FROM sales_data
WHERE category = 'K-Beauty'
GROUP BY 1;

#02 체화 재고 리스크 분석 SQL

-- 1년 이상 장기 미판매 SKU 추출
SELECT 
    sku_id, 
    last_sale_date,
    stock_qty,
    DATEDIFF('2026-02-12', last_sale_date) as aging_days
FROM inventory
WHERE stock_qty > 0 
  AND last_sale_date < DATE_SUB(NOW(), INTERVAL 1 YEAR);
